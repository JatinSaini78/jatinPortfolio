Under the hood

How HydraPet is built

SwiftUI views talk to @Observable view models. The rules live in small services, and data sits behind repositories, so the app can run on SwiftData and the tests on in-memory stores. The code below is taken straight from the project.

  • 01

    A day is its own type

    Logs, streaks and challenges use a CalendarDay in the user's time zone instead of a raw Date, so a drink logged just before midnight can't land on the wrong day.

  • 02

    Today never breaks a streak

    Only an unmet yesterday resets the streak, and the goal counts once per day. Both rules live in one small, pure StreakService.

  • 03

    Time is injected, so it can be tested

    An AppClock is passed in rather than read from the system. Tests move it past midnight and check the rollover, across 90 Swift Testing tests.

import Foundation

/// Owns the retention mechanic (PRD §5.3, FR-HOME-09/11).
///
/// Two rules matter most and are easy to get wrong:
///
/// 1. **An in-progress day can never break a streak.** Only *yesterday* being unmet
///    breaks it. A user at 10am with an empty glass still holds their streak.
/// 2. **The goal counts once per day.** Every later log on a day already met is a no-op
///    for the streak, so logging ten times does not add ten days.
nonisolated struct StreakService: Sendable {

    func evaluate(
        currentStreak: Int,
        longestStreak: Int,
        lastGoalMetDate: CalendarDay?,
        lastOpenedDate: CalendarDay?,
        today: CalendarDay,
        isGoalMetToday: Bool,
        wasGoalMetYesterday: Bool
    ) -> StreakState {
        var days = currentStreak
        var longest = longestStreak
        var broken = false
        let alreadyCountedToday = lastGoalMetDate == today

        // Break detection — first foreground of a new day with yesterday unmet.
        if !alreadyCountedToday, days > 0,
           !wasGoalMetYesterday, lastOpenedDate != today {
            longest = max(longest, days)
            days = 0
            broken = true
        }

        // Increment — today's goal has just been met for the first time.
        if isGoalMetToday, !alreadyCountedToday {
            days += 1
            broken = false
            longest = max(longest, days)
        }

        return StreakState(days: days, longestStreak: longest, isBroken: broken)
    }
}

import Foundation

/// A calendar date in the user's local timezone, stored as `yyyy-MM-dd`.
///
/// PRD §5.3: "A DAY = a calendar date in the USER'S LOCAL TIMEZONE." Streaks, challenge
/// progress and midnight transitions all key off this type rather than `Date`, so an
/// instant near midnight can never be attributed to the wrong day.
///
/// `yyyy-MM-dd` is zero-padded and fixed-width, so lexicographic ordering is also
/// chronological ordering — which is why `Comparable` can simply compare raw strings.
nonisolated struct CalendarDay: Hashable, Comparable, Codable, Sendable {
    let rawValue: String

    private static func formatter(for timeZone: TimeZone) -> DateFormatter {
        let formatter = DateFormatter()
        formatter.locale = Locale(identifier: "en_US_POSIX")
        formatter.dateFormat = "yyyy-MM-dd"
        formatter.timeZone = timeZone
        return formatter
    }

    init(date: Date, calendar: Calendar) {
        rawValue = Self.formatter(for: calendar.timeZone).string(from: date)
    }

    init?(rawValue: String) {
        guard Self.formatter(for: TimeZone(identifier: "UTC") ?? .gmt)
            .date(from: rawValue) != nil else { return nil }
        self.rawValue = rawValue
    }

    func date(in calendar: Calendar) -> Date {
        Self.formatter(for: calendar.timeZone).date(from: rawValue) ?? Date()
    }

    func adding(days: Int, calendar: Calendar) -> CalendarDay {
        let base = date(in: calendar)
        let shifted = calendar.date(byAdding: .day, value: days, to: base) ?? base
        return CalendarDay(date: shifted, calendar: calendar)
    }

    /// Whole days from `self` to `other`. Negative when `other` is earlier.
    func days(until other: CalendarDay) -> Int {
        var utc = Calendar(identifier: .gregorian)
        utc.timeZone = TimeZone(identifier: "UTC") ?? .gmt
        return utc.dateComponents([.day],
                                  from: date(in: utc),
                                  to: other.date(in: utc)).day ?? 0
    }

    static func < (lhs: CalendarDay, rhs: CalendarDay) -> Bool {
        lhs.rawValue < rhs.rawValue
    }
}

import Testing
@testable import HydraPet

// From HydraPetTests/SmokeTests.swift
@Test("an in-progress day never breaks a streak")
func inProgressDayIsSafe() {
    let today = CalendarDay(rawValue: "2025-09-15")!
    let yesterday = CalendarDay(rawValue: "2025-09-14")!
    let state = StreakService().evaluate(
        currentStreak: 6, longestStreak: 6,
        lastGoalMetDate: yesterday, lastOpenedDate: yesterday, today: today,
        isGoalMetToday: false, wasGoalMetYesterday: true)
    #expect(state.days == 6)
    #expect(state.isBroken == false)
}
